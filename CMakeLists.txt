cmake_minimum_required(VERSION 3.20)
project(OCRAITranslator VERSION 0.9.1 LANGUAGES CXX RC)
set(SAT_VERSION "${PROJECT_VERSION}-preview")
configure_file(src/app_version.h.in generated/app_version.h @ONLY)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "OCR AI Translator requires Windows and MSVC (x64).")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Configure with -A x64.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(UILayoutTests tests/ui_layout_tests.cpp)
target_include_directories(UILayoutTests PRIVATE src)
add_executable(Probe src/probe.cpp src/ocr_adapter.cpp)
add_executable(SpikeRunner src/spike_runner.cpp)
add_executable(ClipboardTests tests/native_clipboard_tests.cpp)
add_executable(CoreTests tests/core_tests.cpp tests/config_review_tests.cpp src/config_store.cpp src/sse_parser.cpp src/translation_client.cpp src/ocr_adapter.cpp src/provider_diagnostics.cpp)
add_executable(ProcessTests tests/process_tests.cpp)
add_executable(SnipasteLaunchFixture tests/snipaste_launch_fixture.cpp)
target_include_directories(SnipasteLaunchFixture PRIVATE src)
add_dependencies(CoreTests SnipasteLaunchFixture)
add_dependencies(ProcessTests SnipasteLaunchFixture)
add_executable(Translator src/translator.cpp src/app_resources.rc src/config_store.cpp src/translation_client.cpp src/sse_parser.cpp src/ocr_adapter.cpp src/provider_diagnostics.cpp)
add_executable(TranslatorTest src/translator.cpp src/app_resources.rc src/config_store.cpp src/translation_client.cpp src/sse_parser.cpp src/ocr_adapter.cpp src/provider_diagnostics.cpp)
set_property(SOURCE src/app_resources.rc APPEND PROPERTY OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/app-icon/translator.ico")
foreach(app Translator TranslatorTest)
  target_include_directories(${app} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  add_custom_command(TARGET ${app} POST_BUILD
    COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/scripts/refresh-app-icons.ps1" -Executable "$<TARGET_FILE:${app}>"
    COMMENT "Refresh the application file icon in Windows Shell"
    VERBATIM)
endforeach()
set_target_properties(Translator TranslatorTest PROPERTIES WIN32_EXECUTABLE TRUE)
target_include_directories(ClipboardTests PRIVATE src)
target_include_directories(CoreTests PRIVATE src)
target_compile_definitions(CoreTests PRIVATE SAT_ENABLE_TEST_HOOKS=1)
target_include_directories(ProcessTests PRIVATE src)
target_include_directories(Translator PRIVATE src)
target_include_directories(TranslatorTest PRIVATE src)
target_compile_definitions(TranslatorTest PRIVATE SAT_ENABLE_TEST_HOOKS=1)
foreach(target UILayoutTests Probe SpikeRunner ClipboardTests CoreTests ProcessTests Translator TranslatorTest SnipasteLaunchFixture)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_compile_options(${target} PRIVATE /W4 /WX /utf-8 /permissive- /EHsc)
  target_link_libraries(${target} PRIVATE user32 gdi32 bcrypt shell32 advapi32 dwmapi crypt32 winhttp comctl32 shlwapi ole32)
  target_link_options(${target} PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/probe.manifest)
endforeach()
target_link_libraries(CoreTests PRIVATE ws2_32)
target_link_libraries(ProcessTests PRIVATE user32 ws2_32)
target_link_libraries(Translator PRIVATE imm32 comdlg32 d2d1)
target_link_libraries(TranslatorTest PRIVATE imm32 comdlg32 d2d1)
enable_testing()
add_test(NAME ui_layout_geometry COMMAND UILayoutTests)
add_test(NAME probe_unit COMMAND Probe --self-test)
add_test(NAME native_clipboard_isolated COMMAND ClipboardTests)
add_test(NAME core_tests COMMAND CoreTests)
add_test(NAME process_lifecycle COMMAND ProcessTests $<TARGET_FILE:TranslatorTest>)
add_test(NAME translator_gui_smoke COMMAND TranslatorTest --self-test-ui)
add_test(NAME capture_window_lifecycle COMMAND ProcessTests $<TARGET_FILE:TranslatorTest> --capture-window-tests)
add_test(NAME ui_stability_faults COMMAND TranslatorTest --stability-test-ui)
set_tests_properties(ui_stability_faults PROPERTIES TIMEOUT 180 ENVIRONMENT "SAT_TEST_MODE=1;SAT_CONFIG_DIR=${CMAKE_CURRENT_BINARY_DIR}/stability-ui-config" RESOURCE_LOCK translator_test_instance)
add_test(NAME network_stability_faults COMMAND CoreTests --stability)
set_tests_properties(network_stability_faults PROPERTIES TIMEOUT 180 ENVIRONMENT "SAT_TEST_MODE=1;SAT_CONFIG_DIR=${CMAKE_CURRENT_BINARY_DIR}/test-config" RESOURCE_LOCK core_test_config)
set_tests_properties(core_tests PROPERTIES RESOURCE_LOCK core_test_config)
add_test(NAME process_stability_faults COMMAND ProcessTests $<TARGET_FILE:TranslatorTest> --stability-tests)
set_tests_properties(process_stability_faults PROPERTIES TIMEOUT 180 ENVIRONMENT "SAT_TEST_MODE=1;SAT_CONFIG_DIR=${CMAKE_CURRENT_BINARY_DIR}/process-test-config" RESOURCE_LOCK translator_test_instance)
set_tests_properties(capture_window_lifecycle PROPERTIES TIMEOUT 45 ENVIRONMENT "SAT_TEST_MODE=1;SAT_CONFIG_DIR=${CMAKE_CURRENT_BINARY_DIR}/capture-test-config" RESOURCE_LOCK translator_test_instance)
set_tests_properties(native_clipboard_isolated PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
set_tests_properties(core_tests PROPERTIES TIMEOUT 30 ENVIRONMENT "SAT_TEST_MODE=1;SAT_CONFIG_DIR=${CMAKE_CURRENT_BINARY_DIR}/test-config")
set_tests_properties(process_lifecycle PROPERTIES TIMEOUT 60 ENVIRONMENT "SAT_TEST_MODE=1;SAT_CONFIG_DIR=${CMAKE_CURRENT_BINARY_DIR}/process-test-config")
set_tests_properties(translator_gui_smoke PROPERTIES TIMEOUT 60 ENVIRONMENT "SAT_TEST_MODE=1;SAT_CONFIG_DIR=${CMAKE_CURRENT_BINARY_DIR}/gui-test-config")
set_tests_properties(process_lifecycle translator_gui_smoke PROPERTIES RESOURCE_LOCK translator_test_instance)
set_tests_properties(ui_layout_geometry probe_unit core_tests network_stability_faults PROPERTIES LABELS "headless;local_fixture")
set_tests_properties(process_lifecycle translator_gui_smoke capture_window_lifecycle ui_stability_faults process_stability_faults PROPERTIES LABELS "native_gui;local_fixture")
set_tests_properties(native_clipboard_isolated PROPERTIES LABELS "clipboard;environment_dependent")
